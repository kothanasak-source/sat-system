    <script>
        let DB = {};
        let currentUser = localStorage.getItem("sat_user") || "";

        // On Page Load
        window.onload = () => {
            if (currentUser) {
                document.getElementById('login-modal').classList.add('hidden');
                document.getElementById('user-badge').innerText = `👤 ${currentUser}`;
                fetchDatabase();
            }
        };

        async function handleLogin() {
            const u = document.getElementById('login-user').value;
            const p = document.getElementById('login-pass').value;
            const err = document.getElementById('login-err');
            
            try {
                const res = await fetch('/api/login', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ username: u, password: p })
                });
                const data = await res.json();
                if (data.success) {
                    localStorage.setItem("sat_user", data.user);
                    currentUser = data.user;
                    document.getElementById('login-modal').classList.add('hidden');
                    document.getElementById('user-badge').innerText = `👤 ${currentUser}`;
                    if (data.is_offline) document.getElementById('offline-banner').classList.remove('hidden');
                    fetchDatabase();
                } else {
                    err.innerText = data.message || "Invalid Login";
                    err.classList.remove('hidden');
                }
            } catch (e) {
                err.innerText = "Connection error";
                err.classList.remove('hidden');
            }
        }

        function logout() {
            localStorage.removeItem("sat_user");
            location.reload();
        }

        async function fetchDatabase() {
            try {
                const res = await fetch('/api/database');
                const result = await res.json();
                if (result.success) {
                    DB = result.data;
                    if (result.is_offline) {
                        document.getElementById('offline-banner').classList.remove('hidden');
                    } else {
                        document.getElementById('offline-banner').classList.add('hidden');
                    }
                    populateFilters();
                }
            } catch (e) {
                document.getElementById('offline-banner').classList.remove('hidden');
            }
        }

        function populateFilters() {
            const dataSheet = DB['Data'] || [];
            if (dataSheet.length < 2) return;
            
            const selTopic = document.getElementById('sel-topic');
            selTopic.innerHTML = "";
            for (let i = 1; i < dataSheet.length; i++) {
                if (dataSheet[i][0]) selTopic.innerHTML += `<option value="${dataSheet[i][0]}">${dataSheet[i][0]}</option>`;
            }
            onTopicChange();
        }

        function onTopicChange() {
            // ดึง Sheet Equipment ตาม Topic
            const selTopic = document.getElementById('sel-topic').value;
            const dataSheet = DB['Data'] || [];
            let eqSheetName = "Equipment";
            for (let i = 1; i < dataSheet.length; i++) {
                if (dataSheet[i][0] === selTopic && dataSheet[i][3]) eqSheetName = dataSheet[i][3];
            }

            const eqRows = DB[eqSheetName] || [];
            const phases = [...new Set(eqRows.slice(1).map(r => r[0]).filter(Boolean))];
            
            const selPhase = document.getElementById('sel-phase');
            selPhase.innerHTML = phases.map(p => `<option value="${p}">${p}</option>`).join('');
            onPhaseChange();
        }

        function onPhaseChange() {
            const selTopic = document.getElementById('sel-topic').value;
            const selPhase = document.getElementById('sel-phase').value;
            const dataSheet = DB['Data'] || [];
            let eqSheetName = "Equipment";
            for (let i = 1; i < dataSheet.length; i++) {
                if (dataSheet[i][0] === selTopic && dataSheet[i][3]) eqSheetName = dataSheet[i][3];
            }

            const eqRows = DB[eqSheetName] || [];
            const rooms = [...new Set(eqRows.slice(1).filter(r => r[0] === selPhase).map(r => r[1]).filter(Boolean))];
            
            const selRoom = document.getElementById('sel-room');
            selRoom.innerHTML = rooms.map(r => `<option value="${r}">${r}</option>`).join('');
            onRoomChange();
        }

        function onRoomChange() {
            renderCabinetsAndSubjects();
        }

        function renderCabinetsAndSubjects() {
            const selTopic = document.getElementById('sel-topic').value;
            const selPhase = document.getElementById('sel-phase').value;
            const selRoom = document.getElementById('sel-room').value;
            
            const dataSheet = DB['Data'] || [];
            let eqSheetName = "Equipment", subjSheetName = "Subject";
            for (let i = 1; i < dataSheet.length; i++) {
                if (dataSheet[i][0] === selTopic) {
                    subjSheetName = dataSheet[i][2] || "Subject";
                    eqSheetName = dataSheet[i][3] || "Equipment";
                }
            }

            const eqRows = DB[eqSheetName] || [];
            const activeCabs = eqRows.slice(1).filter(r => r[0] === selPhase && r[1] === selRoom).map(r => r[2]);
            
            // อัปเดตสถิติตัวเลข Dashboard
            document.getElementById('txt-room-total').innerText = "1";
            document.getElementById('txt-room-done').innerText = "1";
            document.getElementById('txt-cab-total').innerText = activeCabs.length;
            document.getElementById('txt-cab-done').innerText = activeCabs.length;

            // Render ปุ่ม Cabinet
            const btnContainer = document.getElementById('cab-btn-container');
            btnContainer.innerHTML = activeCabs.map((cab, idx) => `
                <button onclick="setActiveCabinet('${cab}')" class="bg-[#0F172A] hover:bg-sky-600 text-sky-400 hover:text-white border border-slate-700 px-3 py-1 rounded text-xs font-bold transition">
                    ${cab.split('-').length >= 3 ? cab.split('-')[cab.split('-').length - 3] : cab}
                </button>
            `).join('');

            // Render Subjects
            const subjRows = DB[subjSheetName] || [];
            const subContainer = document.getElementById('subjects-container');
            subContainer.innerHTML = subjRows.slice(1).map(s => `
                <div class="border border-slate-800 bg-[#0F172A]/60 rounded-lg p-2.5">
                    <div class="text-xs font-extrabold text-sky-300 mb-2">Subject ${s[0]}: ${s[1]}</div>
                    <div class="grid grid-cols-4 gap-2">
                        <label class="border-2 border-dashed border-sky-800 hover:border-sky-500 rounded-lg flex flex-col items-center justify-center p-3 cursor-pointer transition">
                            <span class="text-xs text-sky-400 font-bold">+ Upload S${s[0]}</span>
                            <input type="file" multiple class="hidden" onchange="handleFileUpload(event, ${s[0]})">
                        </label>
                    </div>
                </div>
            `).join('');
        }

        async function handleFileUpload(event, subjectNum) {
            const files = event.target.files;
            if (!files.length) return;
            
            const selTopic = document.getElementById('sel-topic').value;
            const selPhase = document.getElementById('sel-phase').value;
            const selRoom = document.getElementById('sel-room').value;
            
            for (let file of files) {
                const fd = new FormData();
                fd.append("topic", selTopic);
                fd.append("phase", selPhase);
                fd.append("room", selRoom);
                fd.append("eq", "A1");
                fd.append("subject", subjectNum);
                fd.append("file", file);
                
                await fetch('/api/upload', { method: 'POST', body: fd });
            }
            alert(`Uploaded ${files.length} photos successfully!`);
        }

        async function exportReport(type) {
            alert(`กำลังประมวลผลรายงาน ${type.toUpperCase()}... ไฟล์จะเริ่มดาวน์โหลดโดยอัตโนมัติ`);
            const selTopic = document.getElementById('sel-topic').value;
            const selPhase = document.getElementById('sel-phase').value;
            const selRoom = document.getElementById('sel-room').value;
            
            const payload = {
                topic: selTopic, phase: selPhase, room: selRoom,
                test_date: document.getElementById('test-date').value,
                revision: document.getElementById('revision').value,
                user: currentUser
            };

            const res = await fetch('/api/export/word', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (res.ok) {
                const blob = await res.blob();
                const url = window.URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = `SAT_Report_${selTopic}_${selRoom}.docx`;
                document.body.appendChild(a);
                a.click();
                a.remove();
            } else {
                alert("Export Failed: Template not found or data error");
            }
        }
    </script>
